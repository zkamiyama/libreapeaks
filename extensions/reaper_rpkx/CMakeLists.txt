cmake_minimum_required(VERSION 3.20)
project(reaper_rpkx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REAPER_SDK "" CACHE PATH "Path to pinned reaper-sdk repository")
set(WDL_ROOT "" CACHE PATH "Path to WDL repository (contains WDL/swell)")
set(BRIDGE_LIBRARY "" CACHE FILEPATH "Rust static library")
option(LRPK_ENABLE_TEST_HOOKS "Diagnostic build only; never distribute" OFF)
if(NOT EXISTS "${REAPER_SDK}/sdk/reaper_plugin.h")
  message(FATAL_ERROR "REAPER_SDK must contain sdk/reaper_plugin.h")
endif()
add_library(reaper_rpkx SHARED plugin.cpp)
target_include_directories(reaper_rpkx PRIVATE "${REAPER_SDK}/sdk" "${WDL_ROOT}/WDL")
target_link_libraries(reaper_rpkx PRIVATE "${BRIDGE_LIBRARY}")
set_target_properties(reaper_rpkx PROPERTIES PREFIX "")
if(LRPK_ENABLE_TEST_HOOKS)
  target_compile_definitions(reaper_rpkx PRIVATE LRPK_TEST_HOOKS=1)
endif()
if(WIN32)
  target_compile_definitions(reaper_rpkx PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(reaper_rpkx PRIVATE ws2_32 userenv bcrypt ntdll)
  target_compile_options(reaper_rpkx PRIVATE /W4 /EHsc)
elseif(APPLE)
  target_link_libraries(reaper_rpkx PRIVATE "-framework Security" "-framework CoreFoundation")
  target_compile_options(reaper_rpkx PRIVATE -Wall -Wextra -Wpedantic)
else()
  target_link_libraries(reaper_rpkx PRIVATE pthread dl m)
  target_compile_options(reaper_rpkx PRIVATE -Wall -Wextra -Wpedantic)
endif()
